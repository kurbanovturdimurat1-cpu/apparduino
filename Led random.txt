int led1=2;
int led2=3;
int led3=4;

void setup() {
  pinMode(led1,OUTPUT);
  pinMode(led2,OUTPUT);
  pinMode(led3,OUTPUT);
  
  Serial.begin(9600);

  randomSeed(analogRead(0));

}

void loop() {
  int tanlapin=random(2,5);
  int tanlavaqt=random(100,500);
  digitalWrite(tanlapin, HIGH);
  Serial.println("Tanlangan pin:"+String(tanlapin));
  Serial.println("___________________________________________");
  delay(tanlavaqt);
  Serial.print("Vaqt: "+String(tanlavaqt));
  Serial.println("____________________________________________");
  digitalWrite(tanlapin, LOW);
  delay(tanlavaqt);


}
